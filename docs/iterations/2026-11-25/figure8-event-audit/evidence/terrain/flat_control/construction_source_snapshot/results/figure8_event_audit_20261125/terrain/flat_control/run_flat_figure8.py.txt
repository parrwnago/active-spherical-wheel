"""I41 single flat-environment counterfactual of the saved I40 Mars reference.

Reuses the exact preserved I40 logging/guarded native-controller runner. Local
construction wrappers supply a constant-height 500x500 mesh with the original
Mars DEM domain, vertex order, diagonal and output precision. No controller or
sealed I40 source is modified. This removes global relief and obstacles; it
does not isolate one slope or preserve the original initial physical state.
"""
from pathlib import Path
from types import SimpleNamespace
from unittest.mock import patch
from datetime import datetime, timezone
import hashlib
import importlib.util
import json
import shutil

import numpy as np

from spherical_wheel.model import REPO_ROOT, load_config
from spherical_wheel.terrain import Terrain
from spherical_wheel.gazebo.experiments import GazeboWorkspace
from spherical_wheel.gazebo.mars import route_bounds, write_dem_mesh
import spherical_wheel.gazebo.plant as gz_plant
import spherical_wheel.gazebo.worlds as gz_worlds


HERE = Path(__file__).resolve().parent
FAMILY = HERE.parents[1]
OLD = REPO_ROOT/'results/lqr_contact_forecast_20261125/campaign/runs/mars_figure8_p090'
HELPER = OLD/'source_snapshot/scripts/run-lqr-contact-forecast.py'


def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def main():
    protocol = FAMILY/'protocol.json'
    if not protocol.is_file():
        raise FileNotFoundError('Declare I41 protocol before a new Gazebo run')
    directory = HERE/'campaign/runs/flat_figure8'
    if directory.exists():
        raise FileExistsError('Preserve completed/failed counterfactual; no overwrite')
    origin = json.loads((OLD/'provenance.json').read_text())
    with np.load(OLD/'rollout.npz') as z:
        height = float(z['ground_height_m'][0])
    terrain_spec = json.loads((OLD/'terrain.json').read_text())
    count, size = int(terrain_spec['samples']), float(terrain_spec['size_m'])
    spacing = size/(count-1)
    bounds = (-size/2, size/2, -size/2, size/2)
    visual = route_bounds(['plain_figure8'])
    config_path = OLD/'source_snapshot/config/model.yaml'
    config = load_config(config_path)
    job = {'case_id': 'flat_figure8', 'terrain': 'flat_counterfactual', 'phase_rad': .9,
        'path': {'kind': 'mars_route', 'route': 'plain_figure8'}, 'duration_s': 44.,
        'yaw_deg': 45., 'reanchor_at_motion_start': False, 'uniform_anchor_start_s': .4}
    spec = importlib.util.spec_from_file_location('i41_preserved_i40_runner', HELPER)
    runner = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(runner)
    construction = {'iteration': 'I-41', 'declared_before_run_utc': datetime.now(timezone.utc).isoformat(),
        'scope': 'One new Gazebo/DART flat-environment case; old Mars trace reused. No controller or sealed I40 source edit.',
        'job': job, 'height_m': height, 'height_source': 'Saved Mars initial ground_height_m float32; construction only, never passed as a contact observation.',
        'physical_mesh_bounds': bounds, 'physical_mesh_vertices': [count, count], 'physical_mesh_spacing_m': spacing,
        'physical_mesh_method': 'Same write_dem_mesh vertex order/diagonal/6decimal OBJ formatting on a constant height array.',
        'visual_bounds': visual, 'visual_is_separate_from_physical_domain': True,
        'friction': config['contact']['friction'], 'initialization': 'Same standard terrain-appropriate balanced initialization, nominal phase and yaw. Initial pose/observer differ because terrain differs.',
        'counterfactual_limit': 'Removes relief and all rocks/lava globally, not an isolated small-slope manipulation or state-matched causal experiment.',
        'source_sha256': {str(p.relative_to(REPO_ROOT)): sha(p) for p in
            (Path(__file__), HELPER, protocol, config_path, OLD/'rollout.npz', OLD/'precision.npz', OLD/'provenance.json', OLD/'terrain.json',
             REPO_ROOT/'src/spherical_wheel/gazebo/worlds.py', REPO_ROOT/'src/spherical_wheel/gazebo/plant.py', REPO_ROOT/'src/spherical_wheel/gazebo/mars.py')}}
    snapshot = HERE/'construction_source_snapshot'
    snapshot.mkdir(parents=True, exist_ok=False)
    for p in (Path(__file__), HELPER, protocol, config_path):
        target = snapshot/p.relative_to(REPO_ROOT)
        target.parent.mkdir(parents=True, exist_ok=True)
        shutil.copyfile(p, target)
    (HERE/'construction.json').write_text(json.dumps(construction, indent=2)+'\n')
    (HERE/'campaign').mkdir(exist_ok=True)
    (HERE/'campaign/jobs.json').write_text(json.dumps({'iteration': 'I-41', 'cases': [job]}, indent=2)+'\n')
    shutil.copyfile(protocol, HERE/'protocol.json')
    flat_terrain = Terrain(name='flat_counterfactual', features=[{'type': 'plane', 'offset': height, 'slope': [0., 0.]}])

    def workspace(*args, **kwargs):
        kwargs['heightmap_cell_m'] = spacing
        kwargs['ground'] = 'mesh'
        options = dict(kwargs.get('plant_options', {}))
        options['visual_bounds'] = visual
        kwargs['plant_options'] = options
        return GazeboWorkspace(*args, **kwargs)

    real_world = gz_plant.write_world

    def world(directory, configuration=None, **kwargs):
        kwargs.update(mesh_bounds=bounds, cell_m=spacing)
        return real_world(directory, configuration, **kwargs)

    def mesh(terrain, mesh_bounds, cell_m, path):
        assert tuple(mesh_bounds) == bounds and np.isclose(cell_m, spacing)
        dummy = SimpleNamespace(count=count, size=size, spacing=spacing,
                                heights=np.full((count, count), height))
        write_dem_mesh(dummy, path)
        return count, count, spacing, spacing, height, height

    with patch.object(runner, 'Terrain', lambda *a, **k: flat_terrain), \
         patch.object(runner, 'GazeboWorkspace', workspace), \
         patch.object(gz_plant, 'write_world', world), \
         patch.object(gz_worlds, 'write_terrain_mesh', mesh):
        result = runner.simulate(job, HERE)
    (HERE/'result.json').write_text(json.dumps(result, indent=2)+'\n')
    print(json.dumps({'result': result, 'construction': construction}, indent=2), flush=True)


if __name__ == '__main__':
    main()
