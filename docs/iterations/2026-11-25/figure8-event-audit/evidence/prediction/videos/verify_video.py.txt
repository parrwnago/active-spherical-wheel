"""Decode and verify the I41 saved-state presentation without running physics."""
from pathlib import Path
import hashlib
import json
import shutil
import imageio.v2 as imageio
import numpy as np

HERE = Path(__file__).resolve().parent
ROOT = HERE.parents[3]


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main():
    if (HERE/'qa.json').exists():
        raise FileExistsError('Preserve completed QA stage')
    source=Path(__file__).resolve()
    snapshot=HERE/'source_snapshot'/f'verify_video_{sha(source)[:12]}.py'
    shutil.copy2(source,snapshot)
    metadata=json.loads((HERE/'video_metadata.json').read_text())
    target=ROOT/metadata['video']['path']
    assert sha(target)==metadata['video']['sha256']
    assert sha(HERE/'render_figure8_events.py')==metadata['render_source_sha256']
    assert sha(ROOT/metadata['render_source_snapshot'])==metadata['render_source_sha256']
    for name,expected in metadata['input_source_sha256'].items():
        assert sha(ROOT/name)==expected,name
    for name,expected in metadata['stills'].items():
        assert sha(HERE/name)==expected,name
    assert metadata['reference_and_time_arrays_exact_between_actual_cases']
    assert metadata['chart_axes_common_between_actual_cases']
    assert metadata['playback_speed']==.5 and len(metadata['chapters'])==3
    for chapter in metadata['chapters']:
        np.testing.assert_allclose(np.asarray(chapter['display_window_s'])-chapter['time_s'],[-1,2],atol=1e-12)
        assert max(abs(np.asarray(chapter['actual_endpoint_times_s'])-chapter['display_window_s']))<=.0025
    selected={40:'chapter01_event',161:'chapter02_event',282:'chapter03_event',422:'endcard'}
    decoded=[]
    with imageio.get_reader(target) as reader:
        info=reader.get_meta_data()
        for index,frame in enumerate(reader):
            assert frame.shape==(920,1920,3) and np.isfinite(frame).all()
            assert np.std(frame)>5
            if index in selected:
                path=HERE/f'qa_decoded_{selected[index]}.jpg'
                imageio.imwrite(path,frame,quality=92)
                decoded.append({'index':index,'path':path.name,'sha256':sha(path)})
        count=index+1
    assert count==423==metadata['video']['frames']
    assert info['fps']==20 and abs(info['duration']-21.15)<.001
    output={'passed':True,'scope':'All-frame saved MP4 decode and exact source/input/still hashes; no physics, controller or forecast execution.',
        'video':metadata['video'],'render_source_sha256':metadata['render_source_sha256'],
        'QA_source_sha256':sha(source),'QA_source_snapshot':str(snapshot.relative_to(ROOT)),
        'all_423_frames_decoded':True,'all_frames_nonblank_and_expected_dimensions':True,
        'input_hashes_verified':len(metadata['input_source_sha256']),
        'exact_reference_and_program_timestamps_between_actual_cases':True,
        'matched_event_windows':True,'maximum_endpoint_sampling_offset_s':.0025,
        'decoded_evaluation_frames':decoded,
        'labels_reviewed':['Actual Gazebo/DART state replay rendered with MuJoCo; no new dynamics run.',
            'Mars actual DEM and flat full-terrain counterfactual; balanced initial pose also differs.',
            'Same native LQR and active kinematic synchronizer; shadow forecasts never actuated.',
            'First turn lag is distinct from geometric overshoot; later turn06 actual overshoot is shown.',
            'B uses FUTURE actual inputs only as explicit offline hindsight diagnostic.',
            'Endcard onset anchor passes initial load fit; five poor fits remain included.'],
        'development_stages_preserved':['development/stage01_preview_boundary_failure','development/stage02_initial_endcard'],
        'saved_data_prediction_QA':'../saved_data_QA/report.json'}
    (HERE/'qa.json').write_text(json.dumps(output,indent=2)+'\n')
    print(json.dumps(output,indent=2))


if __name__=='__main__':
    main()
