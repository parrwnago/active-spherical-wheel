def sha(path):return hashlib.sha256(Path(path).read_bytes()).hexdigest()
def load_npz(path):
    with np.load(path) as z:return {k:z[k].copy() for k in z.files}
def safe(value):
    if isinstance(value,np.ndarray):return safe(value.tolist())
    if isinstance(value,np.generic):return safe(value.item())
    if isinstance(value,dict):return {str(k):safe(v) for k,v in value.items()}
    if isinstance(value,(tuple,list)):return [safe(v) for v in value]
    if isinstance(value,float) and not np.isfinite(value):return None
    return value
def save(path,value):path.write_text(json.dumps(safe(value),indent=2,allow_nan=False)+'\n')


def matrix(quaternions):
    q=np.asarray(quaternions);w,x,y,z=q.T
    return np.stack((1-2*(y*y+z*z),2*(x*y-w*z),2*(x*z+w*y),
        2*(x*y+w*z),1-2*(x*x+z*z),2*(y*z-w*x),
        2*(x*z-w*y),2*(y*z+w*x),1-2*(x*x+y*y)),axis=1).reshape(-1,3,3)


def lean_reconstruction(raw):
    actual=matrix(raw['exact_qpos'][:,3:7])[:,:,2]
    result={}
    for label,key in [('raw','exact_raw_reference_up_error_rad'),('shaped','exact_shaped_reference_up_error_rad')]:
        desired=matrix(raw[f'tilt_{label}_reference_quat'])[:,:,2]
        # atan2 is independently stable near zero and avoids acos cancellation.
        reconstructed=np.arctan2(np.linalg.norm(np.cross(actual,desired),axis=1),np.einsum('ij,ij->i',actual,desired))
        result[f'{label}_up_error_max_gap_rad']=float(np.max(abs(reconstructed-raw[key])))
    result['actual_up_max_gap']=float(np.max(abs(actual-raw['exact_actual_up'])))
    result['passed']=max(result.values())<1e-6
    return result


def reference_lift_audit(raw):
    raw_lift=Rotation.from_rotvec(np.column_stack((raw['tilt_raw_theta_rad'],np.zeros(len(raw['time_s']))))).as_matrix()
    shaped_lift=Rotation.from_rotvec(np.column_stack((raw['tilt_theta_rad'],np.zeros(len(raw['time_s']))))).as_matrix()
    raw_rotation=matrix(raw['tilt_raw_reference_quat']);shaped_rotation=matrix(raw['tilt_shaped_reference_quat'])
    anchors=np.einsum('nij,nkj->nik',raw_rotation,raw_lift)
    anchor=anchors[0]
    reconstructed=np.einsum('ij,njk->nik',anchor,shaped_lift)
    world=np.einsum('ij,nj->ni',anchor,raw['tilt_angular_velocity_equilibrium_rad_s'])
    result={'equilibrium_anchor_max_matrix_variation':float(abs(anchors-anchor).max()),
        'shaped_lift_max_matrix_error':float(abs(reconstructed-shaped_rotation).max()),
        'equilibrium_to_world_rate_max_gap_rad_s':float(abs(world-raw['tilt_angular_velocity_world_rad_s']).max())}
    result['passed']=max(result.values())<1e-12
    return result


def common_prefix(base,other):
    states=('time_s','exact_qpos','exact_qvel','exact_previous_applied','exact_observer_wrench','exact_estimated_normal',
            'exact_nominal_joints_before','native_state_fingerprint_before','comparable_state_fingerprint_before')
    commands=('exact_native_requested','exact_applied','native_state_fingerprint_after')
    fields={}
    for key in states+commands:
        # First active command deliberately emits exactly native quaternion/zero rate.
        ma=base['time_s']<=4.+1e-9;mb=other['time_s']<=4.+1e-9
        fields[key]=equality(base[key][ma],other[key][mb])
    n=min(len(base['time_s']),len(other['time_s']))
    reference={key:equality(base[key][:n],other[key][:n]) for key in (
        'time_s','exact_reference_xy','exact_reference_velocity_m_s','exact_reference_acceleration_m_s2')}
    return {'all_prefix_fields_bit_equal':all(x['bit_equal'] for x in fields.values()),'prefix_fields':fields,
        'all_common_reference_fields_bit_equal':all(x['bit_equal'] for x in reference.values()),'reference_fields':reference,
        'common_reference_samples':n,'arm_samples':len(other['time_s']),
        'full_fingerprint_bookkeeping_difference_is_expected':'Enabled generator tracks preactivation clock; comparison uses canonical dynamic state. True full hash is not required equal.',
        'true_full_before_warm_comparison':equality(base['full_state_fingerprint_before'][base['time_s']<=4.+1e-9],
            other['full_state_fingerprint_before'][other['time_s']<=4.+1e-9])}


def generator_audit(arm,folder,raw,provenance,independent):
    record=next(x for x in provenance['sources'] if x['origin']=='src/spherical_wheel/tilt_reference.py')
    module_path=folder/record['snapshot'];assert sha(module_path)==record['sha256']
    spec=importlib.util.spec_from_file_location('i43_'+arm,module_path);module=importlib.util.module_from_spec(spec)
    sys.modules[spec.name]=module;spec.loader.exec_module(module)
    result={'module_sha256':record['sha256'],'matches_independent_tested_module':record['sha256']==independent,
        'cadence_max_gap_s':float(np.max(abs(np.diff(raw['time_s'])-.005))) if len(raw['time_s'])>1 else None}
    if arm=='baseline':
        result.update(inactive_all_rows=not bool(raw['tilt_active'].any()),used_rate_max_abs=float(abs(raw['tilt_used_angular_velocity_body_rad_s']).max()))
        result['auxiliary_clocks_correctly_absent']=bool(np.isnan(raw['tilt_clock_before_s']).all() and np.isnan(raw['tilt_clock_after_s']).all())
        result['passed']=result['matches_independent_tested_module'] and result['inactive_all_rows'] and result['used_rate_max_abs']==0. and result['auxiliary_clocks_correctly_absent'] and result['cadence_max_gap_s'] is not None and result['cadence_max_gap_s']<1e-10
        return result
    settings=module.TiltReferenceSettings(**provenance['parameters']['tilt_reference'])
    generator=module.TiltReference(settings,np.radians(provenance['parameters']['max_lean_deg']))
    keys=('theta_rad','z_rad','held_raw_theta_rad','theta_dot_rad_s','theta_ddot_rad_s2','angular_velocity_equilibrium_rad_s')
    gaps={key:0. for key in keys};flags=True;max_h=0.
    for i,t in enumerate(raw['time_s']):
        row=generator.update(raw['tilt_raw_theta_rad'][i],float(t))
        for key in keys:gaps[key]=max(gaps[key],float(abs(row[key]-raw['tilt_'+key][i]).max()))
        flags=flags and all(row[key]==raw['tilt_'+key][i] for key in ('active','initialized_now','advanced','substeps'))
        if row['advanced']:max_h=max(max_h,(t-raw['time_s'][i-1])/row['substeps'])
    active=raw['tilt_active'].astype(bool);limit=np.radians(provenance['parameters']['max_lean_deg']);rate=np.radians(settings.max_rate_deg_s)
    bounds={'theta_disk_max_rad':float(np.linalg.norm(raw['tilt_theta_rad'][active],axis=1).max()),
        'z_disk_max_rad':float(np.linalg.norm(raw['tilt_z_rad'][active],axis=1).max()),
        'chart_rate_max_rad_s':float(np.linalg.norm(raw['tilt_theta_dot_rad_s'][active],axis=1).max()),
        'chart_acceleration_max_rad_s2':float(np.linalg.norm(raw['tilt_theta_ddot_rad_s2'][active],axis=1).max()),
        'world_rate_max_rad_s':float(np.linalg.norm(raw['tilt_angular_velocity_world_rad_s'][active],axis=1).max()),
        'max_integrator_substep_s':float(max_h)} if active.any() else None
    actual=matrix(raw['exact_qpos'][:,3:7])
    transported=np.einsum('nji,nj->ni',actual,raw['tilt_angular_velocity_world_rad_s'])
    body_gap=float(abs(transported-raw['tilt_angular_velocity_body_rad_s']).max())
    used=raw['tilt_angular_velocity_body_rad_s'] if settings.use_rate_reference else np.zeros_like(transported)
    used_gap=float(abs(used-raw['tilt_used_angular_velocity_body_rad_s']).max())
    feedback_gap=float(abs(raw['exact_feedback_reference_qvel'][:,3:6]-raw['tilt_used_angular_velocity_body_rad_s']).max())
    bounds_pass=bool(bounds is not None and bounds['theta_disk_max_rad']<=limit+1e-12 and bounds['z_disk_max_rad']<=limit+1e-12
        and bounds['chart_rate_max_rad_s']<=rate+1e-12 and bounds['chart_acceleration_max_rad_s2']<=2*rate/settings.time_constant_s+1e-10
        and bounds['world_rate_max_rad_s']<=rate+1e-12 and max_h<=min(.001,settings.time_constant_s/4)+1e-12)
    result.update({'replayed_saved_raw_input_max_gaps':gaps,'replayed_flags_equal':bool(flags),'bounds':bounds,'bounds_pass':bounds_pass,
        'actual_body_transport_max_gap_rad_s':body_gap,'used_rate_max_gap_rad_s':used_gap,'feedback_rate_max_gap_rad_s':feedback_gap})
    result['auxiliary_clock_before_matches_previous_call']=bool(np.array_equal(raw['tilt_clock_before_s'],np.r_[np.nan,raw['time_s'][:-1]],equal_nan=True))
    result['auxiliary_clock_after_matches_current_call']=bool(np.array_equal(raw['tilt_clock_after_s'],raw['time_s']))
    result['passed']=bool(result['matches_independent_tested_module'] and flags and max(gaps.values())==0. and bounds_pass
        and max(body_gap,used_gap,feedback_gap)<1e-12 and result['cadence_max_gap_s'] is not None and result['cadence_max_gap_s']<1e-10
        and result['auxiliary_clock_before_matches_previous_call'] and result['auxiliary_clock_after_matches_current_call'])
    return result


def window_metrics(raw,log,geometry,bounds,window):
    result=summarize(raw['time_s'],raw,log,geometry,bounds,window)
    mask=observation_window(raw['time_s'],*window)
    for name,key in [('raw','exact_raw_reference_up_error_rad'),('shaped','exact_shaped_reference_up_error_rad'),('absolute','exact_absolute_tilt_rad')]:
        values=np.degrees(raw[key][mask])
        result[name+'_tilt_RMS_deg']=float(np.sqrt(np.mean(values**2))) if len(values) else None
        result[name+'_tilt_max_deg']=float(values.max()) if len(values) else None
    result['belt_A_max_abs_requested_m_s']=float(abs(raw['exact_native_requested'][mask,3]).max()) if mask.any() else None
    return result


def turns(raw,geometry,events):
    t=raw['time_s'];rows=[]
    for event in events:
        at=event['time_s'];lo,hi=event['window_s'];mask=observation_window(t,lo,hi);after=mask&(t>=at)
        axis=event['axis_index'];signed=event['incoming_sign']*(raw['exact_qpos'][:,axis]-event['reference_component_extremum_m'])
        candidates=actual_reversal_candidates(t,raw['exact_qvel'][:,axis],event['incoming_sign'],event['outgoing_sign'],lo,hi,.1)
        rows.append({'event_id':event['event_id'],'time_s':at,'axis':event['axis'],'window_s':[lo,hi],
            'window_complete':bool(t[0]<=lo and t[-1]>=hi),'geometric_component_overshoot_mm':max(0.,float(1000*signed[mask].max())) if mask.any() else None,
            'actual_reversal_lag_s':candidates[0]['interpolated_zero_time_s']-at if candidates else None,
            'actual_reversal_missing':not bool(candidates),'actual_reversal_multiple':len(candidates)>1,'actual_reversal_candidates':candidates,
            'post_peak_local_error_mm':float(1000*geometry['local_path_distance_m'][after].max()) if after.any() else None})
    return rows
