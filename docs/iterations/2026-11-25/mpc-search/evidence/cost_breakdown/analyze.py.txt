"""Decompose already-saved frozen costs; no solver or dynamics calls."""
from pathlib import Path
import json,csv,hashlib
import numpy as np
HERE=Path(__file__).resolve().parent
FAMILY=HERE.parent
REPO=FAMILY.parents[1]
OLD=REPO/'results/feedback_recovery_20261125/runs/slope/mpc/plans.json'
if (HERE/'report.json').exists():raise FileExistsError('Preserve earlier breakdown')
paths=[Path(__file__),OLD,FAMILY/'frozen_search/anchor_0.4/native_rollout.json',FAMILY/'frozen_search/anchor_2/native_rollout.json']
(HERE/'protocol.json').write_text(json.dumps({'scope':'Read-only decomposition of saved nonlinear states and exact existing objective; no new physics, optimization or tuning.','anchors_s':[.4,2.],'inputs_sha256':{str(p.relative_to(REPO)):hashlib.sha256(p.read_bytes()).hexdigest() for p in paths}},indent=2)+'\n')
plans=json.loads(OLD.read_text());rows=[]
scales=np.array([.05,.05,.005,.10,.10,.5,1,1,1,.10,.10,.1,1,1,1,10,10,10])
weights=np.array([2,2,.1,50,50,0,0,0,0,2,2,.05,5,5,.01,.005,.005,.005])
groups={'xy_position':(0,1),'normal_height':(2,),'tilt':(3,4),'unpenalized_angles':(5,6,7,8),'linear_velocity':(9,10,11),'angular_velocity':(12,13,14),'internal_velocity':(15,16,17)}
for at in (.4,2.):
 p=min(plans,key=lambda x:abs(x['time_s']-at));target=np.array(p['target_states']);T=np.eye(18);T[2,:3]=p['normal'];T[11,9:12]=p['normal']
 native=json.loads((FAMILY/f'frozen_search/anchor_{at:g}/native_rollout.json').read_text())['history'][0]
 candidates=[('saved_selected_mpc',p['nonlinear_states'],p['nonlinear_cost'],p['cost_components'],float(np.linalg.norm(np.array(p['qpos'])[:2]+np.array(p['nonlinear_states'])[-1,:2]-np.array(p['qpos'])[:2]-target[-1,:2])*1000)),
             ('predicted_native_seed',native['nonlinear_states'],native['cost'],native['cost_components'],native['predicted_terminal_goal_error_mm'])]
 for name,states,total,components,terminal in candidates:
  residual=(np.array(states)-target)@T.T/scales
  costs=residual**2*weights;costs[-1]*=4
  row={'anchor_s':at,'candidate':name,'total_cost':total,'predicted_terminal_goal_error_mm':terminal,
       **{key:float(costs[:,list(index)].sum()) for key,index in groups.items()},
       'input':components['input_cost'],'input_change':components['change_cost']}
  reconstructed=sum(row[k] for k in groups)+row['input']+row['input_change']
  assert np.isclose(reconstructed,total,rtol=1e-12,atol=1e-10),(reconstructed,total)
  row['reconstructed_total']=reconstructed;rows.append(row)
report={'scope':'Saved-cost component comparison, not new candidates or controller runs.','rows':rows,
        'limitation':'A short-horizon cost breakdown does not prove a weight/horizon fix or a global optimum; predicted native seed is a private-model policy, not actual future LQR commands.'}
(HERE/'report.json').write_text(json.dumps(report,indent=2)+'\n')
with (HERE/'metrics.csv').open('w',newline='') as f:
 w=csv.DictWriter(f,fieldnames=list(rows[0]));w.writeheader();w.writerows(rows)
print(json.dumps(rows,indent=2))
