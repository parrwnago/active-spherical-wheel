"""Package verified I34 evidence and saved-state media; no new experiments."""
from pathlib import Path
import hashlib,json,zipfile
ROOT=Path(__file__).resolve().parents[2];FAMILY=Path(__file__).resolve().parent
DOCS=ROOT/'docs/iterations/2026-11-25/mpc-search'
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
zip_path=FAMILY/'presentation-pack.zip'
if zip_path.exists():raise FileExistsError('Preserve old pack; choose a new packaging stage')
paths=[ROOT/'ITERATION_LOG.md',ROOT/'README.md',ROOT/'OFFROAD.md',DOCS.parent/'README.md',DOCS/'README.md',Path(__file__).resolve(),ROOT/'results/mpc_equilibrium_20261125/independent_search/report.json']
paths += [p for p in (DOCS/'evidence').rglob('*') if p.is_file()]
paths += list((FAMILY/'videos').glob('*.mp4'))
folders=[FAMILY/'runs/slope'/v for v in ('native','mpc')]
folders += [ROOT/'results/feedback_recovery_20261125/runs/slope'/v for v in ('native','mpc')]
for f in folders:paths += [f/'rollout.npz',f/'precision.npz']
paths=sorted(set(paths));entries=[{'path':str(p.relative_to(ROOT)),'bytes':p.stat().st_size,'sha256':sha(p)} for p in paths]
m={'scope':'I34 presentation evidence. Two new Gazebo cases; two I32 baselines reused. Frozen state searches/contact forecasts are private diagnostics.',
 'tests':'124 selected integration checks; standalone15seedchecks overlap, not additive.',
 'model_boundary':'Observer-derived current contact; fixed local estimated plane in private predictions. No actualfuturecontrols orlivecontacttruth supplied toMPC.',
 'raw_selection':'4rollout+4precisionNPZincluded. FullMPCplans/generatedmeshpayloads omittedwithhash/pathinventories.',
 'artifacts':entries,'payload_count':len(entries),'payload_bytes':sum(e['bytes'] for e in entries)}
manifest=FAMILY/'presentation_manifest.json';manifest.write_text(json.dumps(m,indent=2)+'\n')
with zipfile.ZipFile(zip_path,'x',compression=zipfile.ZIP_DEFLATED,compresslevel=6) as z:
 for p in paths:z.write(p,p.relative_to(ROOT))
 z.write(manifest,manifest.relative_to(ROOT))
with zipfile.ZipFile(zip_path) as z:
 assert z.testzip() is None
 for e in entries:assert hashlib.sha256(z.read(e['path'])).hexdigest()==e['sha256'],e['path']
 assert len(z.namelist())==len(entries)+1
v={'zip':str(zip_path.relative_to(ROOT)),'zip_bytes':zip_path.stat().st_size,'zip_sha256':sha(zip_path),'file_count_including_manifest':len(entries)+1,'zip_crc_verified':True,'all_payload_hashes_verified':True}
(FAMILY/'presentation_verification.json').write_text(json.dumps(v,indent=2)+'\n')
zip_path.with_suffix('.zip.sha256').write_text(v['zip_sha256']+'  '+zip_path.name+'\n')
(DOCS/'presentation_manifest.json').write_bytes(manifest.read_bytes());(DOCS/'presentation_verification.json').write_bytes((FAMILY/'presentation_verification.json').read_bytes())
print(json.dumps(v,indent=2))
