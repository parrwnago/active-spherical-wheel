"""The nonlinear acceptance score must agree with the QP's declared objective."""
import importlib.util
from pathlib import Path

import numpy as np
import pytest

from spherical_wheel.predictive_control import AffineMPC, AffineMPCSettings


SPEC = importlib.util.spec_from_file_location(
    'mpc_comparison_harness', Path(__file__).parents[1]/'scripts/run-mpc-comparison.py')
HARNESS = importlib.util.module_from_spec(SPEC)
SPEC.loader.exec_module(HARNESS)


class LinearReplay:
    def __init__(self, a, b, c, anchor):
        self.a, self.b, self.c, self.anchor = a, b, c, anchor

    def rollout(self, q, v, inputs, dt):
        poses = [np.zeros(len(q))]
        for u in inputs:
            poses.append(self.a@poses[-1]+self.b@(u-self.anchor)+self.c)
        return np.asarray(poses), np.zeros_like(poses)

    def tangent(self, q0, v0, q, v):
        return q


@pytest.mark.parametrize('centered', [False, True])
def test_nonlinear_screening_and_qp_use_the_same_augmented_objective(centered):
    settings = AffineMPCSettings(horizon_steps=4, state_scales=(.3, .7),
        state_weights=(2., 3.), input_weight=.2, change_weight=.4,
        eps_abs=1e-9, eps_rel=1e-9)
    a = np.array([[1., .02], [0., .97]])
    b = np.array([[.01], [.15]])
    c, anchor, previous, bounds = np.array([.001, -.002]), np.array([.1]), np.array([.2]), np.array([.8])
    target = np.tile([.015, -.01], (4, 1))
    transform = np.diag([1., 2.])
    inverse = np.linalg.inv(transform)
    parameters = {}
    if centered:
        # A nonzero cross term checks that the last input is normalized and
        # centered in both the condensed QP and nonlinear line-search score.
        factor = np.array([[1., .3, -.7], [.1, .8, .2]])
        parameters = dict(target_inputs=np.tile([.12], (4, 1)),
            terminal_weight=np.diag([4., 5.]), terminal_tail_weight=factor.T@factor)
    result = AffineMPC(settings).solve(transform@a@inverse, transform@b, transform@c,
        target@transform.T, anchor, previous, bounds, **parameters)
    assert result.success, result.diagnostics
    experiment = object.__new__(HARNESS.TrajectoryPredictiveExperiment)
    experiment.settings = settings
    experiment.predictor = LinearReplay(a, b, c, anchor)
    states, score, components = experiment.evaluate(np.zeros(2), np.zeros(2),
        result.inputs, target, transform, bounds, previous, **parameters)
    np.testing.assert_allclose(states@transform.T, result.states, atol=1e-11)
    regularization = settings.regularization*np.sum((result.inputs/bounds)**2)
    assert score+regularization == pytest.approx(result.diagnostics['objective'], rel=1e-10, abs=1e-10)
    assert score == pytest.approx(sum(components.values()))
