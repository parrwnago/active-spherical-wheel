"""Visualize saved first private-model audit; does not run dynamics."""
from pathlib import Path
import json
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

ROOT=Path(__file__).resolve().parent
first=json.loads((ROOT/'audit/first_y2.json').read_text())
phase=json.loads((ROOT/'audit/phase_and_inverse_y2.json').read_text())['phase_adjusted_fit']
rows=first['rows']+[phase]
labels=['Analytic pose\nzero input','Analytic pose\nfitted torques','Nearby pose\nfitted torques','Nearby pose + phase\nfitted torques']
fig,axes=plt.subplots(1,2,figsize=(10,4.2),constrained_layout=True)
for ax,key,unit,limit in zip(axes,['hold_xy_mm','hold_rotation_deg'],['XY drift (mm)','Attitude change (deg)'],[.2,.1]):
    values=[row[key] for row in rows]
    ax.bar(range(4),values,color=['#8b8d95','#c77834','#5b86a8','#427e70'])
    ax.set_yscale('log');ax.set_xticks(range(4),labels,fontsize=8)
    ax.axhline(limit,color='#ad2937',linestyle='--',label=f'Declared hold limit: {limit:g}')
    ax.set_ylabel(unit);ax.set_ylim(limit/2,max(values)*2.1)
    for i,v in enumerate(values): ax.text(i,v*1.1,f'{v:.3g}',ha='center',fontsize=9)
    ax.legend(fontsize=8,loc='upper right');ax.grid(axis='y',alpha=.15)
fig.suptitle('I-31: 0.2 s held-input check at saved Y2 s goal reference\nPrivate estimated-plane model; one frozen anchor, original XY goal fixed',fontsize=11)
fig.savefig(ROOT/'audit/first_hold_comparison.png',dpi=180)
fig.savefig(ROOT/'audit/first_hold_comparison.pdf')
