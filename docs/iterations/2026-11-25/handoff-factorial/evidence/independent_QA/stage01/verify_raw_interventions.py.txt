"""Independent I42 saved-array/scope audit. Never imports or runs a controller."""
from pathlib import Path
import argparse
import hashlib
import json
import shutil
import numpy as np

HERE=Path(__file__).resolve().parent
FAMILY=HERE.parent
ROOT=FAMILY.parents[1]
ARMS=('baseline','feedback_shares','sync_off','both')


def read(path):
    return json.loads(path.read_text())


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main():
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--stage',default='stage01')
    args=parser.parse_args()
    out=HERE/args.stage;out.mkdir(exist_ok=False)
    shutil.copy2(__file__,out/Path(__file__).name)
    protocol=read(FAMILY/'protocol.json')
    assert protocol['iteration']=='I-42'
    report={'iteration':'I-42','passed':False,
        'scope':'Independent raw saved-array/hash audit only; zero physics, forecasting or controller execution.',
        'source_sha256':sha(Path(__file__)),'protocol_sha256':sha(FAMILY/'protocol.json'),'arms':{}}
    all_arrays={};verified_files=0
    for arm in ARMS:
        directory=FAMILY/'campaign/runs'/arm
        assert (directory/'artifact_manifest.json').is_file(),f'Arm is not sealed: {arm}'
        provenance=read(directory/'provenance.json')
        for record in read(directory/'artifact_manifest.json')['files']:
            path=directory/record['path']
            assert path.stat().st_size==record['bytes'] and sha(path)==record['sha256'],str(path)
            verified_files+=1
        for record in provenance['sources']:
            assert sha(directory/record['snapshot'])==record['sha256']
            assert sha(ROOT/record['origin'])==record['sha256']
        assert 'execution_failure.json' not in {p.name for p in directory.iterdir()}
        with np.load(directory/'precision.npz') as z:
            exact={k:z[k] for k in z.files}
        with np.load(directory/'rollout.npz') as z:
            log={k:z[k] for k in z.files}
        summary=read(directory/'summary.json')
        time=exact['time_s'];active=(time>=4.-1e-9)&(time<10.-1e-9)
        oracle=active & (arm in ('feedback_shares','both'))
        mask=active & (arm in ('sync_off','both'))
        np.testing.assert_allclose(np.diff(time),.005,atol=1e-12,rtol=0)
        np.testing.assert_array_equal(time,exact['truth_time_s'])
        np.testing.assert_array_equal(time,log['time_s'])
        np.testing.assert_array_equal(exact['exact_qpos'],log['qpos'])
        np.testing.assert_array_equal(exact['exact_qvel'],log['qvel'])
        np.testing.assert_array_equal(exact['exact_applied'],log['input'])
        np.testing.assert_array_equal(exact['exact_native_requested'],log['command'])
        np.testing.assert_array_equal(exact['intervention_window_active'],active)
        np.testing.assert_array_equal(exact['oracle_requested'],oracle)
        np.testing.assert_array_equal(exact['sync_mask_active'],mask)
        native=exact['native_feedback_loads_N'];truth=exact['truth_feedback_loads_N'];used=exact['used_feedback_loads_N']
        np.testing.assert_array_equal(truth,exact['truth_patch_loads_N'])
        assert np.isfinite(native).all() and np.isfinite(truth).all() and np.isfinite(used).all()
        assert np.min(native)>=0 and np.min(truth)>=0 and np.min(used)>=0
        inferred_total=native.sum(axis=1);truth_total=truth.sum(axis=1)
        # touching is computed natively from the original measurement total.
        touching=inferred_total>.001
        code=np.ones(len(time),dtype=int)
        code[oracle & ~touching]=2
        code[oracle & touching & (inferred_total<=.001)]=3
        code[oracle & touching & (inferred_total>.001) & (truth_total<=.05)]=4
        eligible=oracle & touching & (inferred_total>.001) & (truth_total>.05)
        code[eligible]=0
        np.testing.assert_array_equal(exact['oracle_fallback_code'],code)
        np.testing.assert_array_equal(exact['oracle_applied'],eligible)
        desired=native.copy()
        desired[eligible]=truth[eligible]/truth_total[eligible,None]*inferred_total[eligible,None]
        np.testing.assert_allclose(used,desired,atol=1e-12,rtol=1e-12)
        np.testing.assert_allclose(used.sum(axis=1),inferred_total,atol=1e-12,rtol=1e-12)
        scope_fields=[k for k in exact if k.startswith('scope_') and k.endswith('_unchanged')]
        assert len(scope_fields)==7 and all(np.all(exact[k]) for k in scope_fields)
        np.testing.assert_array_equal(exact['used_sync_weights'],log['sync_weights'])
        np.testing.assert_array_equal(exact['used_sync_weights'][~mask],exact['native_sync_weights'][~mask])
        assert not np.any(exact['used_sync_weights'][mask])
        # Both returned top-level and per-target weights are independently checked.
        for field in ('blend_weight','target_joint_speed_rad_s','target_surface_speed_m_s','target_torque_Nm',
                      'normal_load_N','normal_load_fraction','lookahead_s','current_patch_margin_m',
                      'predicted_patch_margin_m','predicted_support_gap_m'):
            a=exact[f'native_sync_target_{field}'];b=exact[f'used_sync_target_{field}']
            if field=='blend_weight':
                assert not np.any(np.nan_to_num(b[mask]))
                np.testing.assert_array_equal(a[~mask],b[~mask])
            else:
                np.testing.assert_array_equal(a,b)
        for name in ('native','used'):
            expected=np.zeros((len(time),3))
            expected[:,1:3]=.005*np.nan_to_num(exact[f'{name}_sync_target_blend_weight'][:,:2])*np.nan_to_num(exact[f'{name}_sync_target_target_joint_speed_rad_s'][:,:2])
            np.testing.assert_allclose(exact[f'{name}_sync_nominal_increment'],expected,atol=1e-15,rtol=0)
        assert not np.any(exact['used_sync_nominal_increment'][mask])
        np.testing.assert_array_equal(exact['used_sync_nominal_increment'][~mask],exact['native_sync_nominal_increment'][~mask])
        hook=read(directory/'hook_rows.json')['rows']
        np.testing.assert_array_equal(np.array([r['time_s'] for r in hook]),time[active])
        for row in hook:
            assert row['feedback_calls']==row['preview_calls']==1
            assert all(row['same_state_invariants'].values())
            if row['sync_mask_active']:
                assert not any(row['used_preview']['blend_weights'])
                assert not any(t['blend_weight'] for t in row['used_preview']['targets'])
        bounds=np.asarray(provenance['input_bounds'])
        assert np.isfinite(exact['exact_qpos']).all() and np.isfinite(exact['exact_qvel']).all()
        assert np.isfinite(exact['exact_applied']).all() and np.all(abs(exact['exact_applied'])<=bounds+1e-12)
        complete_window=bool(time[-1]>=10.-1e-9)
        if complete_window:
            assert int(active.sum())==1200
        report['arms'][arm]={'rows':len(time),'completed':summary['completed'],'failure':summary['failure'],
            'last_time_s':float(time[-1]),'full_declared_window_observed':complete_window,'window_holds':int(active.sum()),
            'oracle_requested_holds':int(oracle.sum()),'oracle_applied_holds':int(eligible.sum()),
            'oracle_active_fallback_counts':{str(k):int(np.sum(oracle & (code==k))) for k in (2,3,4)},
            'sync_mask_holds':int(mask.sum()),'scope_fields_all_true':scope_fields,
            'maximum_inferred_load_total_change_N':float(np.max(abs(used.sum(axis=1)-inferred_total))),
            'maximum_used_masked_sync_weight':float(np.max(abs(exact['used_sync_weights'][mask]))) if mask.any() else None,
            'maximum_used_masked_joint_prep_increment_rad':float(np.max(abs(exact['used_sync_nominal_increment'][mask]))) if mask.any() else None,
            'source_hashes_verified':len(provenance['sources']),'precision_sha256':sha(directory/'precision.npz'),
            'provenance_sha256':sha(directory/'provenance.json')}
        all_arrays[arm]=exact
    baseline=all_arrays['baseline']
    prefix_fields=('exact_qpos','exact_qvel','exact_observer_wrench','exact_observer_generalized',
        'exact_observer_interval_mean','exact_estimated_normal','exact_nominal_joints_before',
        'exact_reference_xy','exact_reference_velocity_m_s','exact_reference_acceleration_m_s2')
    for arm,exact in all_arrays.items():
        state_prefix=exact['time_s']<=4.+1e-9
        command_prefix=exact['time_s']<4.-1e-9
        for field in prefix_fields:
            np.testing.assert_array_equal(exact[field][state_prefix],baseline[field][state_prefix])
        np.testing.assert_array_equal(exact['controller_fingerprint_before'][state_prefix],baseline['controller_fingerprint_before'][state_prefix])
        for field in ('exact_applied','exact_native_requested','controller_fingerprint_after'):
            np.testing.assert_array_equal(exact[field][command_prefix],baseline[field][command_prefix])
        report['arms'][arm]['prefix_states_observer_normals_nominal_references_including4s_exact']=True
        report['arms'][arm]['prefix_commands_and_full_history_fingerprints_before4s_exact']=True
    report.update(passed=True,verified_artifact_files=verified_files,all_arms_independently_checked=True)
    report['limits']=['Used feedback load vectors are call arguments; native dropout can reuse retained _held gains/loads.',
        'Effects can persist after10s through physical/controller histories; direct truth routing and masks stop.',
        'Current raw truth is the last1ms solved force, not filtered or interval-averaged.',
        'Inputs are bounded by existing simulation assumptions, not measured hardware limits.',
        'QA completion is separate from trajectory completion or adoption screening.']
    (out/'report.json').write_text(json.dumps(report,indent=2)+'\n')
    print(json.dumps(report,indent=2))


if __name__=='__main__':
    main()
