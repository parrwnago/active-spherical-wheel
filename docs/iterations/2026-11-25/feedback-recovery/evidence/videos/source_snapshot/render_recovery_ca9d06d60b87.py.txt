"""Render saved I32 Gazebo recovery states; no controller or physics is run.

MUJOCO_GL=egl OPENBLAS_NUM_THREADS=1 OMP_NUM_THREADS=1 .venv/bin/python -u results/feedback_recovery_20261125/videos/render_recovery.py
Use --preview-native to inspect the native endpoint without requiring MPC data.
Use --native-only to render the full native recovery while the MPC run completes.
"""
from pathlib import Path
import argparse
import hashlib
import json

import imageio.v2 as imageio
import mujoco
import numpy as np
from PIL import Image, ImageDraw
import yaml

from spherical_wheel.offroad import TerrainPlant
from spherical_wheel.offroad_view import _camera, _caption, _draw, _font
from spherical_wheel.terrain import Terrain

HERE=Path(__file__).resolve().parent
FAMILY=HERE.parent
ROOT=HERE.parents[2]
WIDTH,VIEW_HEIGHT,CHART_HEIGHT,FPS,SPEED=960,720,240,20,.5
TARGET=HERE/'feedback_recovery_native_vs_mpc.mp4'
COLORS={'native':(64,181,241,255),'mpc':(238,153,59,255)}


def sha(path):return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def angle_from_vertical(quaternion):
    norm=np.sum(quaternion**2,axis=1)
    return np.degrees(np.arccos(np.clip(1-2*np.sum(quaternion[:,1:3]**2,axis=1)/norm,-1,1)))


def load(variant):
    folder=FAMILY/'runs/slope'/variant
    for name in ('summary.json','provenance.json','precision.npz','rollout.npz','terrain.json','plans.json'):
        if not (folder/name).is_file():raise FileNotFoundError(f'Run not complete: {folder/name}')
    summary=json.loads((folder/'summary.json').read_text())
    provenance=json.loads((folder/'provenance.json').read_text())
    with np.load(folder/'rollout.npz') as z:log={k:z[k].copy() for k in z.files}
    with np.load(folder/'precision.npz') as z:precise={k:z[k].copy() for k in z.files}
    for key,value in provenance['generated_sha256'].items():
        assert sha(folder/key)==value,(variant,key,'payload changed')
    log['time_s']=precise['time_s'];log['qpos']=precise['exact_qpos'];log['qvel']=precise['exact_qvel']
    log['input']=precise['exact_applied'];log['reference_xy']=precise['recovery_goal_xy']
    log['tilt_world_deg']=angle_from_vertical(log['qpos'][:,3:7])
    log['reference_tilt_deg']=angle_from_vertical(log['reference_quat'])
    log['goal_error_mm']=1000*np.linalg.norm(log['qpos'][:,:2]-log['reference_xy'],axis=1)
    log['xy_speed_mm_s']=1000*np.linalg.norm(log['qvel'][:,:2],axis=1)
    log['reference_error_deg']=np.degrees(log['attitude_error_rad'])
    assert np.array_equal(log['reference_xy'],np.tile(log['reference_xy'][0],(len(log['time_s']),1)))
    return {'variant':variant,'folder':folder,'log':log,'precise':precise,'summary':summary,
            'provenance':provenance,'metrics':summary['feedback_recovery']}


def nearest(times,moment):
    i=min(int(np.searchsorted(times,moment)),len(times)-1)
    return i-1 if i and abs(times[i-1]-moment)<=abs(times[i]-moment) else i


def chart_strip(run,index,moment,end,error_limit,tilt_limit):
    log=run['log'];t=log['time_s'];image=Image.new('RGB',(WIDTH,CHART_HEIGHT),(12,22,32))
    draw=ImageDraw.Draw(image,'RGBA');color=COLORS[run['variant']]
    specs=[(18,455,'Distance from fixed goal (mm)','goal_error_mm',error_limit,5.),
           (495,445,'Platform normal from vertical (deg)','tilt_world_deg',tilt_limit,None)]
    for left,width,title,key,limit,threshold in specs:
        top,height=40,160
        def point(x,y):return left+40+(width-50)*float(x)/end,top+height-height*float(y)/limit
        draw.text((left,7),title,font=_font(14),fill=(230,240,247))
        draw.rectangle([point(0,limit),point(.4,0)],fill=(125,138,151,40))
        for frac in (0,.5,1):
            yy=limit*frac
            draw.line([point(0,yy),point(end,yy)],fill=(165,184,201,50),width=1)
            draw.text((left+2,point(0,yy)[1]-7),f'{yy:g}',font=_font(11),fill=(195,212,224))
        for tick in (0,.4,2,4,6,8):
            if tick>end:continue
            p=point(tick,0)
            draw.line([(p[0],top),(p[0],top+height)],fill=(150,175,192,35),width=1)
            draw.text((p[0]-8,top+height+3),f'{tick:g}',font=_font(11),fill=(192,210,221))
        if threshold is not None:
            y=point(0,threshold)[1]
            draw.line([(point(0,0)[0],y),(point(end,0)[0],y)],fill=(166,199,142,190),width=1)
            draw.text((left+width-83,y-15),'5 mm radius',font=_font(10),fill=(188,218,170))
        if index:
            draw.line([point(x,y) for x,y in zip(t[:index+1],log[key][:index+1])],fill=color,width=3)
            if key=='tilt_world_deg':
                draw.line([point(x,y) for x,y in zip(t[:index+1],log['reference_tilt_deg'][:index+1])],fill=(245,203,93,255),width=2)
        current=min(moment,end)
        x=point(current,0)[0]
        draw.line([(x,top),(x,top+height)],fill=(245,246,247,140),width=1)
        p=point(t[index],log[key][index]);draw.ellipse((p[0]-3,p[1]-3,p[0]+3,p[1]+3),fill=color)
        if key=='tilt_world_deg':
            draw.text((left+43,CHART_HEIGHT-20),'Color: actual tilt | gold: reference | 0 deg: level platform',font=_font(10),fill=(210,226,237))
        else:
            draw.text((left+43,CHART_HEIGHT-20),'Gray: fixed-input drift | vertical line: current simulation time',font=_font(10),fill=(210,226,237))
    return np.asarray(image)


def panel(plant,renderer,run,moment,option,end,error_limit,tilt_limit):
    log=run['log'];index=nearest(log['time_s'],moment);now=float(log['time_s'][index])
    stopped=moment>now+.005 and now<float(run['summary']['requested_duration_s'])-1e-6
    active=now>=.4-1e-9;variant=run['variant'];metrics=run['metrics']
    if stopped:stage=f"RUN STOPPED at {now:.3f}s; state held for comparison | {run['summary']['failure']}"
    elif not active:stage='FEEDBACK OFF: fitted fixed torques; native observer advances as shadow'
    elif variant=='native':stage='FEEDBACK ON: existing LQR + contact feedforward + synchronization'
    elif bool(run['precise']['mpc_fallback'][index]):stage='FEEDBACK ON: NATIVE LQR FALLBACK (MPC unavailable at this update)'
    else:stage='FEEDBACK ON: current trajectory MPC; bounded physical input held between replans'
    final=moment>=end-1e-8
    if final:
        passed=bool(run['summary']['completed'] and metrics['final_1s_within_goal_and_speed_and_reference_lean'])
        settled='Final 1s return gate: '+('PASS' if passed else 'FAIL')+' (position <=5mm; XY speed <=5mm/s; reference lean error <=2deg)'
    else:settled='Return requires final full second <=5mm, <=5mm/s and <=2deg reference error.'
    u=log['input'][index]
    lines=[stage,
        f"t {now:.3f}s | goal error {log['goal_error_mm'][index]:.3f}mm | XY speed {log['xy_speed_mm_s'][index]:.2f}mm/s",
        f"ABSOLUTE platform tilt {log['tilt_world_deg'][index]:.2f}deg | reference {log['reference_tilt_deg'][index]:.2f}deg | relative error {log['reference_error_deg'][index]:.3f}deg",
        'Torque [shaft,A,B] Nm: '+np.array2string(u[:3],precision=3)+' | belts [A,B] m/s: '+np.array2string(u[3:],precision=3),
        f"Normal load {float(log['normal_force_N'][index]):.2f}N | actual command near input limit: {bool(np.any(abs(u)>=[1.485,.594,.594,.297,.297]))}",
        'Uniform 10deg side slope (.04m mesh), not Mars | original XY goal fixed; no reset at 0.4s switch.',
        'Level platform = 0deg; current balance reference deliberately leans about 9.73deg on this slope.',settled]
    position=log['qpos'][index,:3]
    camera=_camera(np.r_[position[:2],.22],112.,-25.,1.32)
    frame=_draw(plant,renderer,log,index,camera,option,False)
    heading='GAZEBO / DART: '+('existing LQR + feedforward' if variant=='native' else 'current trajectory MPC (offline prototype)')
    frame=_caption(frame,heading,lines,'Saved Gazebo/DART dynamics reconstructed in MuJoCo | camera follows robot | 0.5x playback | no new physics')
    return np.concatenate([frame,chart_strip(run,index,moment,end,error_limit,tilt_limit)],axis=0)


def main():
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--preview-native',action='store_true')
    parser.add_argument('--native-only',action='store_true')
    args=parser.parse_args()
    single=args.preview_native or args.native_only
    target=HERE/'native_feedback_recovery.mp4' if args.native_only else TARGET
    prefix='native_recovery' if args.native_only else 'recovery'
    if not args.preview_native and target.exists():raise FileExistsError('Preserve existing video; use a new stage')
    source=Path(__file__).resolve();source_hash=sha(source)
    snap=HERE/'source_snapshot'/f'render_recovery_{source_hash[:12]}.py'
    snap.parent.mkdir(exist_ok=True)
    if not snap.exists():snap.write_bytes(source.read_bytes())
    runs=[load('native')] if single else [load(v) for v in ('native','mpc')]
    if len(runs)==2:
        a,b=runs;ma=a['log']['time_s']<.4-1e-9;mb=b['log']['time_s']<.4-1e-9
        for key in ('time_s','qpos','qvel','input','reference_xy'):
            np.testing.assert_array_equal(a['log'][key][ma],b['log'][key][mb])
        for key in ('qpos','qvel'):
            np.testing.assert_array_equal(a['log'][key][nearest(a['log']['time_s'],.4)],b['log'][key][nearest(b['log']['time_s'],.4)])
    end=max(float(r['summary']['requested_duration_s']) for r in runs)
    error_limit=max(10.,np.ceil(max(float(r['log']['goal_error_mm'].max()) for r in runs)/10)*10)
    tilt_limit=max(15.,np.ceil(max(float(r['log']['tilt_world_deg'].max()) for r in runs)/5)*5)
    config_path=runs[0]['folder']/'source_snapshot/config/model.yaml'
    terrain_path=runs[0]['folder']/'terrain.json'
    config=yaml.safe_load(config_path.read_text());terrain=Terrain.from_spec(json.loads(terrain_path.read_text()))
    plant=TerrainPlant(config,terrain,HERE/'_render_scene',visual_bounds=(-.6,.6,-.6,.6))
    option=mujoco.MjvOption();option.geomgroup[3:]=0
    saved=plant.capture()
    times=np.arange(0,end+1e-9,SPEED/FPS)
    assert np.isclose(times[-1],end)
    endpoint_checks=[]
    for run in runs:
        log=run['log'];metrics=run['metrics']
        np.testing.assert_allclose([log['goal_error_mm'][-1],log['xy_speed_mm_s'][-1],log['tilt_world_deg'][-1]],
            [metrics['goal_error_final_mm'],metrics['xy_speed_final_mm_s'],metrics['absolute_platform_tilt_final_deg']],atol=1e-9)
        endpoint_checks.append({'variant':run['variant'],'goal_error_final_mm':float(log['goal_error_mm'][-1]),
            'xy_speed_final_mm_s':float(log['xy_speed_mm_s'][-1]),'absolute_platform_tilt_final_deg':float(log['tilt_world_deg'][-1]),
            'reference_platform_tilt_final_deg':float(log['reference_tilt_deg'][-1]),'matches_saved_report':True,
            'final_1s_gate':bool(run['summary']['completed'] and metrics['final_1s_within_goal_and_speed_and_reference_lean'])})
    try:
        with mujoco.Renderer(plant.model,height=VIEW_HEIGHT,width=WIDTH) as renderer:
            if args.preview_native:
                imageio.imwrite(HERE/'native_endpoint_preview.jpg',panel(plant,renderer,runs[0],end,option,end,error_limit,tilt_limit),quality=92)
                print('Native endpoint preview saved; no physics run.',flush=True)
                return
            def frame(moment):return np.concatenate([panel(plant,renderer,r,moment,option,end,error_limit,tilt_limit) for r in runs],axis=1)
            writer=imageio.get_writer(target,fps=FPS,codec='libx264',quality=8,macro_block_size=2,ffmpeg_log_level='error')
            try:
                for moment in times:writer.append_data(frame(float(moment)))
            finally:writer.close()
            for moment,name in ((0.,prefix+'_first.jpg'),(.4,prefix+'_feedback_on.jpg'),(end,prefix+'_final.jpg')):
                imageio.imwrite(HERE/name,frame(moment),quality=92)
    finally:plant.restore(saved)
    with imageio.get_reader(target) as reader:
        metadata=reader.get_meta_data();frames=reader.count_frames();last=reader.get_data(frames-1)
        imageio.imwrite(HERE/(prefix+'_decoded_final.jpg'),last,quality=92)
    assert frames==len(times) and tuple(metadata['size'])==(WIDTH*len(runs),960) and np.isclose(metadata['fps'],FPS)
    paths=[source,snap,config_path,terrain_path,FAMILY/'protocol.json',
           ROOT/'src/spherical_wheel/offroad_view.py',ROOT/'src/spherical_wheel/offroad.py',ROOT/'src/spherical_wheel/terrain.py']
    for run in runs:paths += [run['folder']/name for name in ('precision.npz','rollout.npz','summary.json','provenance.json','plans.json')]
    record={'status':'rendered_and_decoded_verified','video':{'path':str(target.relative_to(ROOT)),'sha256':sha(target),'bytes':target.stat().st_size,
        'dimensions':list(metadata['size']),'fps':metadata['fps'],'frames':frames,'duration_s':metadata['duration'],
        'simulated_duration_s':end,'playback_speed':SPEED,'final_frame_decoded':True},'source_snapshot':str(snap.relative_to(ROOT)),
        'endpoint_checks':endpoint_checks,'source_artifact_sha256':{str(p.relative_to(ROOT)):sha(p) for p in paths},
        'stills':{p.name:sha(p) for p in (HERE/(prefix+'_first.jpg'),HERE/(prefix+'_feedback_on.jpg'),HERE/(prefix+'_final.jpg'),HERE/(prefix+'_decoded_final.jpg'))},
        'scope':('Single panel' if single else 'Both panels')+' reconstruct actual saved Gazebo/DART recovery dynamics. No private-model dynamics, controller execution or physics during rendering.',
        'timeline':'0-.4s same fixed torques while native observer advances; feedback ON at.4s with no physical/estimator reset; originalXYgoal retained through8.4s.',
        'platform_angles':'Absolute angle of platform normal from gravity vertical;0degmeanslevel. Gold reference intentionallyleansaboutslope; relativeattitudeerror is separate.',
        'terrain':'Uniform10deg side-slope mesh .04m. Not Mars or rough offroad terrain.',
        'visualization':'MuJoCo rendering only; commonphysicalmodel andanalytic10degterrain appearance. Camera follows eachrobot; lowercharts retainfixedgoalmetrics. Stoppedruns holdlaststate with explicit stoppedcaption.',
        'result_caution':'Firstgoalradiusentry isnotsettledreturn. Finalfull1s gates andcontrollerfallback are shown; no broadterrain/hardware/real-time claim.'}
    (HERE/('native_video_metadata.json' if single else 'video_metadata.json')).write_text(json.dumps(record,indent=2)+'\n')
    print(json.dumps(record['video'],indent=2),flush=True)


if __name__=='__main__':main()
