"""Plot saved actual-input replay audit; no dynamics or controller calls."""
from pathlib import Path
import csv
import json
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
import numpy as np

HERE = Path(__file__).resolve().parent
report = json.loads((HERE/'report.json').read_text())
rows = []
for anchor in report['rows']:
    for error in anchor['errors']:
        rows.append({'anchor_s': anchor['anchor_s'], **{key:value for key,value in error.items() if not isinstance(value,list)}})
with (HERE/'metrics.csv').open('w',newline='') as f:
    writer=csv.DictWriter(f,fieldnames=list(rows[0]));writer.writeheader();writer.writerows(rows)
fig,axes=plt.subplots(1,3,figsize=(13,4),constrained_layout=True)
for horizon in report['protocol']['horizons_s']:
    selected=[row for row in rows if row['horizon_s']==horizon]
    label=f'{1000*horizon:g} ms replay'
    for ax,key in zip(axes[:2],('xy_error_mm','rotation_error_deg')):
        ax.plot([row['anchor_s'] for row in selected],[row[key] for row in selected],marker='o',label=label)
for ax,ylabel in zip(axes[:2],('XY prediction mismatch (mm)','Rotation prediction mismatch (deg)')):
    ax.set_yscale('log');ax.set_xlabel('Saved Gazebo anchor (s)');ax.set_ylabel(ylabel);ax.grid(alpha=.2);ax.legend(fontsize=8)
anchors=[row['anchor_s'] for row in report['rows']]
change=[row['selected_plan_forecast']['predicted_distance_change_mm'] for row in report['rows']]
axes[2].bar([f'{a:g}' for a in anchors],change,color=['#be654c' if value>0 else '#338b77' for value in change])
axes[2].axhline(0,color='black',lw=.8);axes[2].set_xlabel('Saved selected plan anchor (s)')
axes[2].set_ylabel('Own forecast goal-distance change (mm)')
axes[2].set_title('Positive = chosen forecast retreats',fontsize=10);axes[2].grid(axis='y',alpha=.2)
fig.suptitle('I-32: actual-input model replay versus selected-plan intent\nSix frozen estimated-plane anchors; reused Gazebo motion, no new controller run',fontsize=11)
fig.savefig(HERE/'replay_audit.png',dpi=180);fig.savefig(HERE/'replay_audit.pdf')
summary={'scope':report['scope'],'aggregates':report['aggregate_over_six_declared_anchors'],
         'forecasts':[{'anchor_s':row['anchor_s'],**row['selected_plan_forecast']} for row in report['rows']],
         'interpretation':'Small replay errors at these six anchors do not support treating plant-model mismatch as the sole explanation. Selected forecasts themselves retreat at some anchors. This does not establish causal attribution, Jacobian validity or a tuning fix.'}
(HERE/'summary.json').write_text(json.dumps(summary,indent=2)+'\n')
print(json.dumps(summary['forecasts'],indent=2))
